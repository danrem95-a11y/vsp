select iq.*, a2.fincatdes, b2.parentname, c2.debetcredit as flag_dk,
a2.fincatdes+' '+b2.parentname+' '+iq.accountcode as is_find
from
(
select
a.fincatcode,b.parentcode,c.accountcode,
isnull(awal.debit,0.00) as awal_debit,
isnull(awal.credit,0.00) as awal_credit,
isnull(mvmt.awal2_debit,0.00) as awal2_debit,
isnull(mvmt.awal2_credit,0.00) as awal2_credit,isnull(mvmt.bln1_debit,0.00) as bln1_debit,
isnull(mvmt.bln1_credit,0.00) as bln1_credit,
isnull(mvmt.bln2_debit,0.00) as bln2_debit,
isnull(mvmt.bln2_credit,0.00) as bln2_credit,
isnull(mvmt.bln3_debit,0.00) as bln3_debit,
isnull(mvmt.bln3_credit,0.00) as bln3_credit,
isnull(mvmt.bln4_debit,0.00) as bln4_debit,
isnull(mvmt.bln4_credit,0.00) as bln4_credit,
isnull(mvmt.bln5_debit,0.00) as bln5_debit,
isnull(mvmt.bln5_credit,0.00) as bln5_credit,
isnull(mvmt.bln6_debit,0.00) as bln6_debit,
isnull(mvmt.bln6_credit,0.00) as bln6_credit,
isnull(mvmt.bln7_debit,0.00) as bln7_debit,
isnull(mvmt.bln7_credit,0.00) as bln7_credit,
isnull(mvmt.bln8_debit,0.00) as bln8_debit,
isnull(mvmt.bln8_credit,0.00) as bln8_credit,
isnull(mvmt.bln9_debit,0.00) as bln9_debit,
isnull(mvmt.bln9_credit,0.00) as bln9_credit,
isnull(mvmt.bln10_debit,0.00) as bln10_debit,
isnull(mvmt.bln10_credit,0.00) as bln10_credit,
isnull(mvmt.bln11_debit,0.00) as bln11_debit,
isnull(mvmt.bln11_credit,0.00) as bln11_credit,
isnull(mvmt.bln12_debit,0.00) as bln12_debit,
isnull(mvmt.bln12_credit,0.00) as bln12_credit
from
gl_cate a,gl_cate_detail b,gl_acc c,
(
	  			SELECT 	gl_balance.AccountCode,
							sum(AmountDebet) as debit,
							sum(AmountCredit) as credit
		 		FROM 	gl_balance
				WHERE 	gl_balance.Period = :arg_tgl1
				GROUP BY gl_balance.AccountCode
				) awal,
(
SELECT gl_journal.account_id,
sum(case when tgl between :arg_tgl1 and :arg_tgl2 then isnull(debet,0) else 0 end) as awal2_debit,
sum(case when tgl between :arg_tgl1 and :arg_tgl2 then isnull(kredit,0) else 0 end) as awal2_credit,
sum(case when tgl between :arg_bln1_awal and :arg_bln1_akhir then isnull(debet,0) else 0 end) as bln1_debit,
sum(case when tgl between :arg_bln1_awal and :arg_bln1_akhir then isnull(kredit,0) else 0 end) as bln1_credit,
sum(case when tgl between :arg_bln2_awal and :arg_bln2_akhir then isnull(debet,0) else 0 end) as bln2_debit,
sum(case when tgl between :arg_bln2_awal and :arg_bln2_akhir then isnull(kredit,0) else 0 end) as bln2_credit,
sum(case when tgl between :arg_bln3_awal and :arg_bln3_akhir then isnull(debet,0) else 0 end) as bln3_debit,
sum(case when tgl between :arg_bln3_awal and :arg_bln3_akhir then isnull(kredit,0) else 0 end) as bln3_credit,
sum(case when tgl between :arg_bln4_awal and :arg_bln4_akhir then isnull(debet,0) else 0 end) as bln4_debit,
sum(case when tgl between :arg_bln4_awal and :arg_bln4_akhir then isnull(kredit,0) else 0 end) as bln4_credit,
sum(case when tgl between :arg_bln5_awal and :arg_bln5_akhir then isnull(debet,0) else 0 end) as bln5_debit,
sum(case when tgl between :arg_bln5_awal and :arg_bln5_akhir then isnull(kredit,0) else 0 end) as bln5_credit,
sum(case when tgl between :arg_bln6_awal and :arg_bln6_akhir then isnull(debet,0) else 0 end) as bln6_debit,
sum(case when tgl between :arg_bln6_awal and :arg_bln6_akhir then isnull(kredit,0) else 0 end) as bln6_credit,
sum(case when tgl between :arg_bln7_awal and :arg_bln7_akhir then isnull(debet,0) else 0 end) as bln7_debit,
sum(case when tgl between :arg_bln7_awal and :arg_bln7_akhir then isnull(kredit,0) else 0 end) as bln7_credit,
sum(case when tgl between :arg_bln8_awal and :arg_bln8_akhir then isnull(debet,0) else 0 end) as bln8_debit,
sum(case when tgl between :arg_bln8_awal and :arg_bln8_akhir then isnull(kredit,0) else 0 end) as bln8_credit,
sum(case when tgl between :arg_bln9_awal and :arg_bln9_akhir then isnull(debet,0) else 0 end) as bln9_debit,
sum(case when tgl between :arg_bln9_awal and :arg_bln9_akhir then isnull(kredit,0) else 0 end) as bln9_credit,
sum(case when tgl between :arg_bln10_awal and :arg_bln10_akhir then isnull(debet,0) else 0 end) as bln10_debit,
sum(case when tgl between :arg_bln10_awal and :arg_bln10_akhir then isnull(kredit,0) else 0 end) as bln10_credit,
sum(case when tgl between :arg_bln11_awal and :arg_bln11_akhir then isnull(debet,0) else 0 end) as bln11_debit,
sum(case when tgl between :arg_bln11_awal and :arg_bln11_akhir then isnull(kredit,0) else 0 end) as bln11_credit,
sum(case when tgl between :arg_bln12_awal and :arg_bln12_akhir then isnull(debet,0) else 0 end) as bln12_debit,
sum(case when tgl between :arg_bln12_awal and :arg_bln12_akhir then isnull(kredit,0) else 0 end) as bln12_credit
FROM gl_journal
WHERE tgl between :arg_tgl1 and :arg_bln12_akhir and
((isnull(gl_journal.show_hide,'1') = '1') or 1 = :arg_show)
GROUP BY gl_journal.account_id
) mvmt
where a.fincatcode = b.fincatcode and
b.parentcode = c.parentcode and
(a.fincatcode = 'IS1001' or a.fincatcode = 'IS1110' or a.fincatcode = 'IS2010' or a.fincatcode = 'IS2110' or a.fincatcode = 'IS2120' or a.fincatcode = 'IS2130' or a.fincatcode = 'IS2230' or a.fincatcode = 'IS2330') and
((isnull(c.show_hide,'1') = '1') or 1 = :arg_show) and
c.accountcode *= awal.accountcode and
c.accountcode *= mvmt.account_id
) iq, gl_cate a2, gl_cate_detail b2, gl_acc c2
where iq.fincatcode = a2.fincatcode and iq.parentcode = b2.parentcode and iq.accountcode = c2.accountcode and
a2.fincatcode = b2.fincatcode and b2.parentcode = c2.parentcode
order by
iq.fincatcode,iq.parentcode,iq.accountcode
